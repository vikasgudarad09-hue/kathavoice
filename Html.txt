  <script>
    pdfjsLib.GlobalWorkerOptions.workerSrc = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';

    // ==========================================
    // BACKEND CONFIGURATION
    // ==========================================
    const BACKEND_URL = 'https://storyvoice-backend.vikasgudarad09.workers.dev';
    const RAZORPAY_KEY_ID = 'rzp_test_YourKeyHere'; // Replace with live Razorpay Key when ready
    const APP_INVITE_URL = window.location.href;

    // Generate unique device ID for secure trial tracking
    let deviceId = localStorage.getItem('sv_device_id');
    if (!deviceId) {
      deviceId = 'dev_' + Math.random().toString(36.substring(2)) + Date.now().toString(36);
      localStorage.setItem('sv_device_id', deviceId);
    }

    let pdfDoc = null;
    let currentPage = 1;
    let currentExplanation = '';
    let serverState = { spotsLeft: 1000, activePrice: 49, trialsRemaining: 2, isPro: false };

    // Fetch initial status from secure Cloudflare Worker backend
    async function syncAppStatus() {
      try {
        const res = await fetch(`${BACKEND_URL}/status?deviceId=${deviceId}`);
        const data = await res.json();
        serverState = data;
        updateUI();
      } catch (err) {
        console.error('Offline mode or backend error');
      }
    }

    function updateUI() {
      document.getElementById('spotsRemaining').textContent = serverState.spotsLeft;
      document.getElementById('modalSpotsLeft').textContent = serverState.spotsLeft;
      document.getElementById('activePriceDisplay').textContent = `₹${serverState.activePrice}`;
      document.getElementById('subscribeNowBtn').textContent = `⚡ Unlock Unlimited for ₹${serverState.activePrice} / month`;

      const badge = document.getElementById('subStatusBadge');
      if (serverState.isPro) {
        badge.textContent = '👑 PRO MEMBER';
        badge.className = 'text-xs font-bold bg-emerald-500/20 border border-emerald-500/40 text-emerald-300 px-3 py-1.5 rounded-full';
      } else {
        badge.textContent = `🎁 ${serverState.trialsRemaining} Free Trial${serverState.trialsRemaining === 1 ? '' : 's'} Left`;
      }
    }

    syncAppStatus();

    document.getElementById('subStatusBadge').onclick = () => {
      if (!serverState.isPro) document.getElementById('paywallModal').classList.remove('hidden');
    };
    document.getElementById('closePaywallBtn').onclick = () => {
      document.getElementById('paywallModal').classList.add('hidden');
    };

    // ==========================================
    // RAZORPAY SUBSCRIPTION CHECKOUT
    // ==========================================
    document.getElementById('subscribeNowBtn').onclick = async () => {
      const options = {
        key: RAZORPAY_KEY_ID,
        amount: serverState.activePrice * 100, // in paise
        currency: 'INR',
        name: 'StoryVoice App',
        description: `Monthly Unlimited Plan (${serverState.activePrice === 49 ? 'First 1k Installers' : 'Standard'})`,
        handler: async function (response) {
          // Notify backend to increment 1,000 counter & mark user Pro
          try {
            const res = await fetch(`${BACKEND_URL}/unlock-pro`, {
              method: 'POST',
              headers: { 'Content-Type': 'application/json' },
              body: JSON.stringify({ deviceId, paymentId: response.razorpay_payment_id || 'test_pay' })
            });
            const data = await res.json();
            serverState.isPro = true;
            serverState.spotsLeft = Math.max(0, serverState.spotsLeft - 1);
            updateUI();
            document.getElementById('paywallModal').classList.add('hidden');
            alert('🎉 Welcome to StoryVoice PRO! Enjoy unlimited human storytelling.');
          } catch (e) {
            alert('Payment received, syncing status...');
            syncAppStatus();
          }
        },
        theme: { color: '#f59e0b' }
      };

      if (RAZORPAY_KEY_ID === 'rzp_test_YourKeyHere') {
        if (confirm(`[TEST MODE] Simulate ₹${serverState.activePrice}/month Razorpay UPI Subscription?`)) {
          const res = await fetch(`${BACKEND_URL}/unlock-pro`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ deviceId, paymentId: 'simulated_payment' })
          });
          serverState.isPro = true;
          updateUI();
          document.getElementById('paywallModal').classList.add('hidden');
          alert('🎉 Simulated Pro Unlocked Successfully!');
        }
        return;
      }

      const rzp = new Razorpay(options);
      rzp.open();
    };

    // ==========================================
    // PDF & DOCX PARSER
    // ==========================================
    const rawText = document.getElementById('rawText');
    document.getElementById('fileInput').onchange = async (e) => {
      const file = e.target.files[0];
      if (!file) return;
      document.getElementById('fileNameDisplay').textContent = `Loaded: ${file.name}`;

      if (file.name.endsWith('.pdf')) {
        const arrayBuffer = await file.arrayBuffer();
        pdfDoc = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
        currentPage = 1;
        document.getElementById('pageControls').classList.remove('hidden');
        await renderPdfPage(currentPage);
      } else if (file.name.endsWith('.docx')) {
        document.getElementById('pageControls').classList.add('hidden');
        const arrayBuffer = await file.arrayBuffer();
        const result = await mammoth.extractRawText({ arrayBuffer });
        rawText.value = result.value.slice(0, 3000);
      } else {
        document.getElementById('pageControls').classList.add('hidden');
        rawText.value = await file.text();
      }
    };

    async function renderPdfPage(num) {
      const page = await pdfDoc.getPage(num);
      const content = await page.getTextContent();
      rawText.value = content.items.map(item => item.str).join(' ');
      document.getElementById('pageIndicator').textContent = `Page ${num} of ${pdfDoc.numPages}`;
    }

    document.getElementById('prevPage').onclick = () => {
      if (pdfDoc && currentPage > 1) renderPdfPage(--currentPage);
    };
    document.getElementById('nextPage').onclick = () => {
      if (pdfDoc && currentPage < pdfDoc.numPages) renderPdfPage(++currentPage);
    };

    // ==========================================
    // MULTI-LANGUAGE HUMAN STORYTELLER ENGINE
    // ==========================================
    document.getElementById('explainBtn').onclick = async () => {
      if (!serverState.isPro && serverState.trialsRemaining <= 0) {
        document.getElementById('paywallModal').classList.remove('hidden');
        return;
      }

      const text = rawText.value.trim();
      if (!text) return alert('Please upload a PDF/book or paste a passage first!');

      const targetLang = document.getElementById('langSelect').value;
      const toneStyle = document.getElementById('styleSelect').value;

      const btn = document.getElementById('explainBtn');
      btn.textContent = `🎧 Preparing Human Explanation in ${targetLang}...`;
      btn.disabled = true;

      try {
        const res = await fetch(`${BACKEND_URL}/explain`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ text, targetLang, toneStyle, deviceId })
        });
        const data = await res.json();

        if (data.error === 'TRIAL_LIMIT_REACHED') {
          serverState.trialsRemaining = 0;
          updateUI();
          document.getElementById('paywallModal').classList.remove('hidden');
          return;
        }

        currentExplanation = data.explanation;
        serverState.trialsRemaining = data.trialsRemaining;
        updateUI();

        document.getElementById('humanScript').textContent = currentExplanation;
        document.getElementById('outputCard').classList.remove('hidden');
        autoSelectLanguageVoice();
        speakText(currentExplanation);
      } catch (err) {
        alert('Could not connect to backend server. Please check your internet.');
      } finally {
        btn.textContent = '✨ Explain Like a Human';
        btn.disabled = false;
      }
    };

    // ==========================================
    // VOICE SYNTHESIS
    // ==========================================
    const voiceSelect = document.getElementById('voiceSelect');

    function loadVoices() {
      const voices = window.speechSynthesis.getVoices();
      voiceSelect.innerHTML = '';
      voices.forEach((v, i) => {
        const opt = document.createElement('option');
        opt.value = i;
        opt.textContent = `${v.name} (${v.lang})`;
        voiceSelect.appendChild(opt);
      });
      autoSelectLanguageVoice();
    }

    function autoSelectLanguageVoice() {
      const voices = window.speechSynthesis.getVoices();
      const langSelect = document.getElementById('langSelect');
      const targetCode = langSelect.options[langSelect.selectedIndex].dataset.lang;
      const shortCode = targetCode.split('-')[0];
      const matchIndex = voices.findIndex(v => v.lang.toLowerCase().includes(shortCode));
      if (matchIndex !== -1) voiceSelect.value = matchIndex;
    }

    document.getElementById('langSelect').onchange = autoSelectLanguageVoice;
    window.speechSynthesis.onvoiceschanged = loadVoices;
    loadVoices();

    function speakText(text) {
      window.speechSynthesis.cancel();
      const utter = new SpeechSynthesisUtterance(text);
      const voices = window.speechSynthesis.getVoices();
      if (voices[voiceSelect.value]) utter.voice = voices[voiceSelect.value];
      utter.rate = 0.94;
      utter.pitch = 1.0;
      window.speechSynthesis.speak(utter);
    }

    document.getElementById('playBtn').onclick = () => speakText(currentExplanation);
    document.getElementById('replayBtn').onclick = () => speakText(currentExplanation);
    document.getElementById('stopBtn').onclick = () => window.speechSynthesis.cancel();

    // ==========================================
    // VIRAL APP-INSTALL SHARE
    // ==========================================
    document.getElementById('viralShareBtn').onclick = async () => {
      const previewSnippet = currentExplanation.slice(0, 140) + '...';
      const shareMessage = `🎧 I just listened to this book explained like a real human on StoryVoice:\n\n"${previewSnippet}"\n\n👉 Install the app to hear the full voice explanation (First 2 explanations are FREE, plus ₹49/mo for the first 1,000 installers!): ${APP_INVITE_URL}`;

      if (navigator.share) {
        await navigator.share({
          title: 'StoryVoice - Hear Books Explained Like a Human',
          text: shareMessage,
          url: APP_INVITE_URL
        });
      } else {
        await navigator.clipboard.writeText(shareMessage);
        alert('Viral App Invite copied! Paste it on WhatsApp to invite friends.');
      }
    };
  </script>
